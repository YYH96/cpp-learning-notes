0909_recursion.cpp+

#include <iostream>
// 전제: 0 <= n <= 12 (이 예제는 int 범위를 고려)
int Factorial(int n) {
    if (n <= 1) return 1;
    return n * Factorial(n - 1);
}
int SumTail(int n, int acc = 0) {
    if (n <= 0) return acc;
    return SumTail(n - 1, acc + n);
}
// 수업의 1, 1, 2, 3, ... 규칙, 전제: 1 <= n <= 46
int Fibonacci(int n) {
    if (n <= 2) return 1;
    int a = 1, b = 1;
    for (int i = 3; i <= n; ++i) {
        int next = a + b;
        a = b;
        b = next;
    }
    return b;
}
int main() {
    std::cout << Factorial(5) << ' ' << SumTail(10) << ' ' << Fibonacci(10) << '\n';
}

1616_find.cpp+

관련 개념: 문자열과 문자열 검색 · 파일 내려받기 ↓

C++
#include <iostream>
#include <string>
int MyFind(const std::string& text, const std::string& need) {
    if (need.empty()) return 0;
    if (need.size() > text.size()) return -1;
    for (std::size_t i = 0; i <= text.size() - need.size(); ++i) {
        std::size_t j = 0;
        while (j < need.size() && text[i + j] == need[j]) ++j;
        if (j == need.size()) return static_cast<int>(i);
    }
    return -1;
}
int main() {
    std::cout << MyFind("sadbutsad", "dbu") << '\n'; // 2
    std::cout << MyFind("abc", "xyz") << '\n';      // -1
}
